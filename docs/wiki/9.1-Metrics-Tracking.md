## Overview

This page provides a comprehensive reference for metrics tracking in Tickflow. Metrics tracking is essential for monitoring application health, user engagement, business performance, and model accuracy. The platform tracks metrics across application performance, business, user engagement, model performance, and infrastructure layers.

## Application Performance Metrics

### API Performance

| Metric | Description | Alert Threshold | Tracking Method |
|--------|-------------|-----------------|-----------------|
| **Response Time (p95)** | 95th percentile of API response time | > 2 seconds | Django middleware, Sentry |
| **Response Time (p99)** | 99th percentile of API response time | > 5 seconds | Django middleware, Sentry |
| **Error Rate** | Percentage of requests returning errors | > 1% | Sentry, Logging |
| **Request Rate (RPS)** | Requests per second | N/A | Django middleware |
| **API Success Rate** | Percentage of successful requests | < 99% | Custom metrics |

### Endpoint-Specific Metrics

| Endpoint | p95 Latency | Error Rate | Daily Requests |
|----------|-------------|------------|----------------|
| `/stock-analysis/` | < 500ms | < 1% | ~1,000 |
| `/technical-indicators/` | < 300ms | < 1% | ~500 |
| `/lstm-predict/` | < 1s | < 2% | ~200 |
| `/news/get-news/` | < 1s | < 2% | ~300 |
| `/prediction-history/` | < 400ms | < 1% | ~400 |

### Implementation

```python
# middleware/metrics.py
import time
import logging
from django.core.cache import cache

logger = logging.getLogger(__name__)

class APIMetricsMiddleware:
    def __call__(self, request):
        start_time = time.time()
        response = self.get_response(request)
        duration = (time.time() - start_time) * 1000
        
        # Track endpoint metrics
        cache_key = f"api_metrics_{request.path}_{request.method}"
        metrics = cache.get(cache_key, {'count': 0, 'total_ms': 0, 'max_ms': 0})
        metrics['count'] += 1
        metrics['total_ms'] += duration
        metrics['max_ms'] = max(metrics['max_ms'], duration)
        cache.set(cache_key, metrics, timeout=3600)
        
        # Log slow requests
        if duration > 2000:
            logger.warning(f"SLOW REQUEST: {request.path} - {duration:.2f}ms")
        
        return response
```

```python
# metrics/business.py
from django.db.models import Count, Q
from datetime import datetime, timedelta
from authentication.models import User, AuditLog
from stocks.models import Prediction

class BusinessMetrics:
    @staticmethod
    def get_user_metrics():
        today = datetime.now().date()
        week_ago = today - timedelta(days=7)
        month_ago = today - timedelta(days=30)
        
        return {
            'daily_active_users': User.objects.filter(last_login__date=today).count(),
            'weekly_active_users': User.objects.filter(last_login__date__gte=week_ago).count(),
            'monthly_active_users': User.objects.filter(last_login__date__gte=month_ago).count(),
            'total_users': User.objects.count(),
            'new_today': User.objects.filter(created_at__date=today).count(),
            'new_this_week': User.objects.filter(created_at__date__gte=week_ago).count(),
            'new_this_month': User.objects.filter(created_at__date__gte=month_ago).count(),
        }
    
    @staticmethod
    def get_prediction_metrics():
        return {
            'total_predictions': Prediction.objects.count(),
            'predictions_today': Prediction.objects.filter(created_at__date=datetime.now().date()).count(),
            'pending_predictions': Prediction.objects.filter(is_correct__isnull=True).count(),
            'resolved_predictions': Prediction.objects.filter(is_correct__isnull=False).count(),
            'correct_predictions': Prediction.objects.filter(is_correct=True).count(),
        }
    
    @staticmethod
    def get_conversion_metrics():
        total = User.objects.count()
        verified = User.objects.filter(email_verified=True).count()
        onboarded = User.objects.filter(onboarded=True).count()
        
        return {
            'email_verification_rate': round((verified / total) * 100, 1) if total > 0 else 0,
            'onboarding_completion_rate': round((onboarded / total) * 100, 1) if total > 0 else 0,
        }
```

```python
# metrics/engagement.py
from datetime import datetime, timedelta
from django.db.models import Count
from authentication.models import User
from stocks.models import Prediction

class UserEngagementMetrics:
    @staticmethod
    def get_retention_rate(cohort_date, day):
        """Calculate retention for a specific cohort."""
        cohort = User.objects.filter(created_at__date=cohort_date)
        total = cohort.count()
        
        if total == 0:
            return 0
        
        retention_date = cohort_date + timedelta(days=day)
        retained = cohort.filter(last_login__date=retention_date).count()
        
        return round((retained / total) * 100, 1)
    
    @staticmethod
    def get_session_metrics():
        # [TODO: Implement session tracking]
        return {
            'avg_session_duration': 0,  # Minutes
            'avg_sessions_per_user': 0,
            'total_sessions': 0,
        }
    
    @staticmethod
    def get_feature_adoption():
        total_users = User.objects.count()
        if total_users == 0:
            return {}
        
        # Users who have made predictions
        users_with_predictions = User.objects.annotate(
            pred_count=Count('predictions')
        ).filter(pred_count__gt=0).count()
        
        # Users who have performed analysis
        users_with_analysis = User.objects.filter(analyses_count__gt=0).count()
        
        return {
            'predictions': round((users_with_predictions / total_users) * 100, 1),
            'analysis': round((users_with_analysis / total_users) * 100, 1),
        }
```

### Implementation

```python
# metrics/model_performance.py
from django.db.models import Avg, Count, Q
from datetime import datetime, timedelta
from stocks.models import Prediction

class ModelPerformanceMetrics:
    @staticmethod
    def get_accuracy_7day():
        seven_days_ago = datetime.now() - timedelta(days=7)
        resolved = Prediction.objects.filter(
            resolution_date__gte=seven_days_ago,
            is_correct__isnull=False
        )
        
        total = resolved.count()
        correct = resolved.filter(is_correct=True).count()
        
        return round((correct / total) * 100, 1) if total > 0 else 0
    
    @staticmethod
    def get_performance_summary():
        return {
            'accuracy_7day': ModelPerformanceMetrics.get_accuracy_7day(),
            'total_predictions': Prediction.objects.count(),
            'pending_predictions': Prediction.objects.filter(is_correct__isnull=True).count(),
            'resolved_predictions': Prediction.objects.filter(is_correct__isnull=False).count(),
            'correct_predictions': Prediction.objects.filter(is_correct=True).count(),
        }
    
    @staticmethod
    def get_per_symbol_performance():
        symbols = Prediction.objects.values_list('stock_symbol', flat=True).distinct()
        results = {}
        
        for symbol in symbols:
            predictions = Prediction.objects.filter(stock_symbol=symbol, is_correct__isnull=False)
            total = predictions.count()
            correct = predictions.filter(is_correct=True).count()
            
            if total > 0:
                results[symbol] = {
                    'accuracy': round((correct / total) * 100, 1),
                    'total': total,
                    'correct': correct,
                }
        
        return results
```

### Implementation

```python
# metrics/infrastructure.py
import redis
from django.core.cache import cache
from django.db import connection

class InfrastructureMetrics:
    @staticmethod
    def get_database_metrics():
        with connection.cursor() as cursor:
            # Get connection count
            cursor.execute("SELECT count(*) FROM pg_stat_activity;")
            connections = cursor.fetchone()[0]
            
            # Get database size
            cursor.execute("SELECT pg_database_size(current_database());")
            size = cursor.fetchone()[0]
        
        return {
            'connections': connections,
            'size_mb': round(size / (1024 * 1024), 2),
        }
    
    @staticmethod
    def get_cache_metrics():
        try:
            if hasattr(cache, 'client'):
                client = cache.client.get_client()
                info = client.info()
                
                return {
                    'connected': True,
                    'used_memory': info.get('used_memory_human', 'N/A'),
                    'hit_rate': InfrastructureMetrics._calculate_hit_rate(info),
                    'keys': info.get('db0', {}).get('keys', 0),
                }
        except Exception:
            pass
        
        return {'connected': False}
    
    @staticmethod
    def _calculate_hit_rate(info):
        hits = info.get('keyspace_hits', 0)
        misses = info.get('keyspace_misses', 0)
        total = hits + misses
        return round((hits / total) * 100, 1) if total > 0 else 0
```

| **Business Metrics** | Total users, Predictions, API usage | Numbers |

### Metrics Dashboard API

```python
# views/metrics_dashboard.py
from rest_framework.views import APIView
from rest_framework.response import Response
from django.core.cache import cache
from datetime import datetime

class MetricsDashboardView(APIView):
    def get(self, request):
        cache_key = "metrics_dashboard"
        cached = cache.get(cache_key)
        if cached:
            return Response(cached)
        
        metrics = {
            'timestamp': datetime.utcnow().isoformat() + 'Z',
            
            # User Metrics
            'users': BusinessMetrics.get_user_metrics(),
            
            # Prediction Metrics
            'predictions': ModelPerformanceMetrics.get_performance_summary(),
            
            # API Metrics
            'api': {
                'requests_today': self.get_api_requests_today(),
                'error_rate': self.get_error_rate(),
                'avg_response_time': self.get_avg_response_time(),
            },
            
            # Infrastructure Metrics
            'infrastructure': InfrastructureMetrics.get_database_metrics(),
            'cache': InfrastructureMetrics.get_cache_metrics(),
            
            # Model Metrics
            'model': {
                'accuracy_7day': ModelPerformanceMetrics.get_accuracy_7day(),
                'f1_7day': self.get_f1_7day(),
            },
        }
        
        cache.set(cache_key, metrics, timeout=60)
        return Response(metrics)
```
