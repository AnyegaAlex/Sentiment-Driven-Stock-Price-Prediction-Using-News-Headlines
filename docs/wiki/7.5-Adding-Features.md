## Step 4: Implementation

### Backend Implementation

#### Step 4.1: Create Model (if needed)

```python
# stocks/models.py
class NewFeature(models.Model):
    user = models.ForeignKey('authentication.User', on_delete=models.CASCADE)
    name = models.CharField(max_length=100)
    data = models.JSONField(default=dict)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        indexes = [
            models.Index(fields=['user', 'created_at']),
        ]

    def __str__(self):
        return f"{self.user.username} - {self.name}"
```

#### Step 4.2: Create Migration

```bash
python manage.py makemigrations
python manage.py migrate
```

#### Step 4.3: Create Serializer

```python
# stocks/serializers.py
from rest_framework import serializers
from .models import NewFeature

class NewFeatureSerializer(serializers.ModelSerializer):
    class Meta:
        model = NewFeature
        fields = ['id', 'user', 'name', 'data', 'created_at', 'updated_at']
        read_only_fields = ['id', 'user', 'created_at', 'updated_at']

    def create(self, validated_data):
        validated_data['user'] = self.context['request'].user
        return super().create(validated_data)
```

#### Step 4.4: Create View

```python
# stocks/views.py
from rest_framework.views import APIView
from rest_framework.response import Response
from rest_framework import status
from rest_framework.permissions import IsAuthenticated
from drf_spectacular.utils import extend_schema
from .models import NewFeature
from .serializers import NewFeatureSerializer

class NewFeatureListView(APIView):
    """List all features for the authenticated user."""
    permission_classes = [IsAuthenticated]

    @extend_schema(
        summary="List user features",
        responses={200: NewFeatureSerializer(many=True)},
        tags=["Features"]
    )
    def get(self, request):
        features = NewFeature.objects.filter(user=request.user).order_by('-created_at')
        serializer = NewFeatureSerializer(features, many=True)
        return Response(serializer.data)

class NewFeatureCreateView(APIView):
    """Create a new feature."""
    permission_classes = [IsAuthenticated]

    @extend_schema(
        summary="Create feature",
        request=NewFeatureSerializer,
        responses={201: NewFeatureSerializer},
        tags=["Features"]
    )
    def post(self, request):
        serializer = NewFeatureSerializer(data=request.data, context={'request': request})
        if serializer.is_valid():
            serializer.save()
            return Response(serializer.data, status=status.HTTP_201_CREATED)
        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)
```

#### Step 4.6: Add Tests

```python
# tests/unit/stocks/test_new_feature.py
import pytest
from django.urls import reverse
from rest_framework.test import APIClient
from rest_framework import status
from authentication.models import User
from stocks.models import NewFeature

@pytest.mark.django_db
class TestNewFeature:
    def setup_method(self):
        self.client = APIClient()
        self.user = User.objects.create_user(username='testuser', password='testpass')
        self.client.force_authenticate(user=self.user)

    def test_create_feature(self):
        """Test creating a new feature."""
        url = reverse('features-create')
        data = {'name': 'Test Feature', 'data': {'key': 'value'}}
        response = self.client.post(url, data, format='json')

        assert response.status_code == status.HTTP_201_CREATED
        assert response.data['name'] == 'Test Feature'
        assert response.data['user'] == self.user.id

    def test_list_features(self):
        """Test listing features."""
        NewFeature.objects.create(user=self.user, name='Feature 1')
        NewFeature.objects.create(user=self.user, name='Feature 2')

        url = reverse('features-list')
        response = self.client.get(url)

        assert response.status_code == status.HTTP_200_OK
        assert len(response.data) == 2
        assert response.data[0]['name'] == 'Feature 2'  # Ordered by -created_at
```

### Frontend Implementation

#### Step 4.1: Create API Client Function

```javascript
// frontend/src/services/api.js
import apiClient from './client';

export const featureApi = {
  list: () => apiClient.get('/features/'),
  create: (data) => apiClient.post('/features/create/', data),
};
```

#### Step 4.2: Create Component

```javascript
// frontend/src/components/features/FeatureList.jsx
import React, { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { featureApi } from '@/services/api';
import { Button } from '@/components/ui/Button';
import { Card, CardContent } from '@/components/ui/Card';

export const FeatureList = () => {
  const [newFeatureName, setNewFeatureName] = useState('');
  const queryClient = useQueryClient();

  const { data: features, isLoading, error } = useQuery({
    queryKey: ['features'],
    queryFn: featureApi.list,
    staleTime: 30 * 1000,
  });

  const createMutation = useMutation({
    mutationFn: (name) => featureApi.create({ name, data: {} }),
    onSuccess: () => {
      queryClient.invalidateQueries(['features']);
      setNewFeatureName('');
    },
  });

  const handleCreate = (e) => {
    e.preventDefault();
    if (newFeatureName.trim()) {
      createMutation.mutate(newFeatureName.trim());
    }
  };

  if (isLoading) return <div>Loading...</div>;
  if (error) return <div>Error loading features</div>;

  return (
    <div>
      <form onSubmit={handleCreate} className="flex gap-2 mb-4">
        <input
          type="text"
          value={newFeatureName}
          onChange={(e) => setNewFeatureName(e.target.value)}
          placeholder="Enter feature name"
          className="border rounded px-3 py-2 flex-1"
        />
        <Button type="submit" disabled={createMutation.isPending}>
          {createMutation.isPending ? 'Creating...' : 'Add Feature'}
        </Button>
      </form>

      <div className="grid gap-2">
        {features?.data?.map((feature) => (
          <Card key={feature.id}>
            <CardContent className="py-3">
              <p className="font-medium">{feature.name}</p>
              <p className="text-sm text-gray-500">
                Created: {new Date(feature.created_at).toLocaleDateString()}
              </p>
            </CardContent>
          </Card>
        ))}
      </div>
    </div>
  );
};
```

```javascript
// frontend/src/__tests__/components/features/FeatureList.test.jsx
import { render, screen, fireEvent } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { FeatureList } from '@/components/features/FeatureList';
import { featureApi } from '@/services/api';

jest.mock('@/services/api');

const queryClient = new QueryClient({
  defaultOptions: { queries: { retry: false } },
});

const wrapper = ({ children }) => (
  <QueryClientProvider client={queryClient}>
    {children}
  </QueryClientProvider>
);

describe('FeatureList', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });

  test('renders features list', async () => {
    const mockFeatures = {
      data: [
        { id: 1, name: 'Feature 1', created_at: '2026-07-20T10:00:00Z' },
        { id: 2, name: 'Feature 2', created_at: '2026-07-21T10:00:00Z' },
      ],
    };
    featureApi.list.mockResolvedValue(mockFeatures);

    render(<FeatureList />, { wrapper });

    expect(await screen.findByText('Feature 1')).toBeInTheDocument();
    expect(await screen.findByText('Feature 2')).toBeInTheDocument();
  });

  test('creates a new feature', async () => {
    featureApi.list.mockResolvedValue({ data: [] });
    featureApi.create.mockResolvedValue({ data: { id: 3, name: 'New Feature' } });

    render(<FeatureList />, { wrapper });

    const input = await screen.findByPlaceholderText('Enter feature name');
    const button = screen.getByText('Add Feature');

    fireEvent.change(input, { target: { value: 'New Feature' } });
    fireEvent.click(button);

    expect(featureApi.create).toHaveBeenCalledWith('New Feature');
  });
});
```

## Step 8: Merge and Deploy

### Merge Process

1. PR is reviewed and approved
2. All tests pass in CI
3. Branch is merged to `develop`
4. Feature is deployed to staging
5. Feature is tested in staging
6. Release to production

| Check | Description |
