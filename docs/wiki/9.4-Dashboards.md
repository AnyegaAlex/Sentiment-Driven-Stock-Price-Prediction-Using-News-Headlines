## Overview

This page provides a comprehensive reference for the monitoring dashboards available in Tickflow. Dashboards provide real-time visibility into application health, business metrics, model performance, and infrastructure status. They are essential for monitoring the platform and responding to issues quickly.

## 2. Business Metrics Dashboard

### Overview

The Business Metrics Dashboard provides visibility into user growth, engagement, and platform usage.

### Panels

| Panel | Metric | Visualization | Update Interval |
|-------|--------|---------------|-----------------|
| **Daily Active Users** | DAU trend | Line chart | 1 hour |
| **Weekly Active Users** | WAU trend | Line chart | 1 hour |
| **Monthly Active Users** | MAU trend | Line chart | 1 hour |
| **New Signups** | Daily signups | Bar chart | 1 hour |
| **Total Users** | Cumulative users | Counter | 1 hour |
| **Predictions Generated** | Daily predictions | Bar chart | 1 hour |
| **Prediction Accuracy** | Accuracy trend | Line chart | 1 day |
| **API Usage** | Daily API calls | Bar chart | 1 hour |

### Implementation

```python
# monitoring/dashboards/business.py
from rest_framework.views import APIView
from rest_framework.response import Response
from django.core.cache import cache
from django.db.models import Count
from datetime import datetime, timedelta
from authentication.models import User
from stocks.models import Prediction

class BusinessDashboardView(APIView):
    def get(self, request):
        """Get business metrics dashboard data."""
        cache_key = "dashboard:business"
        cached = cache.get(cache_key)
        if cached:
            return Response(cached)
        
        data = {
            'timestamp': datetime.utcnow().isoformat() + 'Z',
            'users': self.get_user_metrics(),
            'predictions': self.get_prediction_metrics(),
            'engagement': self.get_engagement_metrics(),
            'conversion': self.get_conversion_metrics(),
        }
        
        cache.set(cache_key, data, timeout=300)
        return Response(data)
    
    def get_user_metrics(self):
        """Get user metrics."""
        today = datetime.now().date()
        week_ago = today - timedelta(days=7)
        month_ago = today - timedelta(days=30)
        
        return {
            'daily_active': User.objects.filter(last_login__date=today).count(),
            'weekly_active': User.objects.filter(last_login__date__gte=week_ago).count(),
            'monthly_active': User.objects.filter(last_login__date__gte=month_ago).count(),
            'total': User.objects.count(),
            'new_today': User.objects.filter(created_at__date=today).count(),
            'new_this_week': User.objects.filter(created_at__date__gte=week_ago).count(),
            'new_this_month': User.objects.filter(created_at__date__gte=month_ago).count(),
        }
    
    def get_prediction_metrics(self):
        """Get prediction metrics."""
        today = datetime.now().date()
        week_ago = today - timedelta(days=7)
        
        return {
            'total': Prediction.objects.count(),
            'today': Prediction.objects.filter(created_at__date=today).count(),
            'this_week': Prediction.objects.filter(created_at__date__gte=week_ago).count(),
            'accuracy': self.get_accuracy(),
            'pending': Prediction.objects.filter(is_correct__isnull=True).count(),
            'resolved': Prediction.objects.filter(is_correct__isnull=False).count(),
        }
    
    def get_conversion_metrics(self):
        """Get conversion metrics."""
        total = User.objects.count()
        verified = User.objects.filter(email_verified=True).count()
        onboarded = User.objects.filter(onboarded=True).count()
        
        return {
            'email_verification_rate': round((verified / total) * 100, 1) if total > 0 else 0,
            'onboarding_completion_rate': round((onboarded / total) * 100, 1) if total > 0 else 0,
        }
```

## 3. Model Performance Dashboard

### Overview

The Model Performance Dashboard provides visibility into the accuracy and health of the prediction models.

### Panels

| Panel | Metric | Visualization | Update Interval |
|-------|--------|---------------|-----------------|
| **Accuracy** | 7-day rolling accuracy | Line chart | 1 day |
| **F1 Score** | 7-day rolling F1 Score | Line chart | 1 day |
| **Prediction Volume** | Daily predictions | Bar chart | 1 day |
| **Per-Symbol Performance** | Accuracy by symbol | Table | 1 day |
| **Drift Detection** | F1 score drift | Alert | 1 day |
| **Confusion Matrix** | Actual vs Predicted | Matrix | 1 day |

### Implementation

```python
# monitoring/dashboards/model.py
from rest_framework.views import APIView
from rest_framework.response import Response
from django.core.cache import cache
from stocks.utils import calculate_performance_metrics, detect_drift
from stocks.models import Prediction

class ModelPerformanceDashboardView(APIView):
    def get(self, request):
        """Get model performance dashboard data."""
        cache_key = "dashboard:model"
        cached = cache.get(cache_key)
        if cached:
            return Response(cached)
        
        data = {
            'timestamp': datetime.utcnow().isoformat() + 'Z',
            'overall': self.get_overall_metrics(),
            'by_symbol': self.get_symbol_metrics(),
            'drift': detect_drift(),
            'confusion_matrix': self.get_confusion_matrix(),
            'trends': self.get_trends(),
        }
        
        cache.set(cache_key, data, timeout=3600)
        return Response(data)
    
    def get_overall_metrics(self):
        """Get overall model performance metrics."""
        predictions = Prediction.objects.filter(is_correct__isnull=False)
        metrics = calculate_performance_metrics(predictions)
        return {
            'accuracy': metrics.get('accuracy', 0),
            'f1': metrics.get('f1', 0),
            'precision': metrics.get('precision', 0),
            'recall': metrics.get('recall', 0),
            'total': predictions.count(),
            'correct': predictions.filter(is_correct=True).count(),
        }
    
    def get_symbol_metrics(self):
        """Get per-symbol performance metrics."""
        symbols = Prediction.objects.values_list('stock_symbol', flat=True).distinct()
        results = {}
        
        for symbol in symbols:
            predictions = Prediction.objects.filter(
                stock_symbol=symbol,
                is_correct__isnull=False
            )
            total = predictions.count()
            correct = predictions.filter(is_correct=True).count()
            
            if total > 0:
                results[symbol] = {
                    'accuracy': round((correct / total) * 100, 1),
                    'total': total,
                    'correct': correct,
                }
        
        return results
    
    def get_confusion_matrix(self):
        """Get confusion matrix."""
        predictions = Prediction.objects.filter(
            is_correct__isnull=False,
            actual_direction__in=['up', 'down']
        )
        
        tp = predictions.filter(
            predicted_movement='up',
            actual_direction='up'
        ).count()
        
        tn = predictions.filter(
            predicted_movement='down',
            actual_direction='down'
        ).count()
        
        fp = predictions.filter(
            predicted_movement='up',
            actual_direction='down'
        ).count()
        
        fn = predictions.filter(
            predicted_movement='down',
            actual_direction='up'
        ).count()
        
        return {'TP': tp, 'FP': fp, 'FN': fn, 'TN': tn}
```

## 4. Infrastructure Dashboard

### Overview

The Infrastructure Dashboard provides visibility into server resources, database health, and cache performance.

### Panels

| Panel | Metric | Visualization | Update Interval |
|-------|--------|---------------|-----------------|
| **CPU Usage** | CPU utilization | Line chart | 1 minute |
| **Memory Usage** | Memory utilization | Line chart | 1 minute |
| **Disk Usage** | Storage utilization | Gauge | 5 minutes |
| **Network I/O** | Incoming/outgoing traffic | Line chart | 1 minute |
| **Database Connections** | Active connections | Gauge | 1 minute |
| **Cache Hit Rate** | Redis hit rate | Gauge | 1 minute |
| **Database Size** | Database size | Counter | 1 hour |

### Implementation

```python
# monitoring/dashboards/infrastructure.py
from rest_framework.views import APIView
from rest_framework.response import Response
from django.core.cache import cache
from django.db import connection

class InfrastructureDashboardView(APIView):
    def get(self, request):
        """Get infrastructure dashboard data."""
        cache_key = "dashboard:infrastructure"
        cached = cache.get(cache_key)
        if cached:
            return Response(cached)
        
        data = {
            'timestamp': datetime.utcnow().isoformat() + 'Z',
            'server': self.get_server_metrics(),
            'database': self.get_database_metrics(),
            'cache': self.get_cache_metrics(),
            'network': self.get_network_metrics(),
        }
        
        cache.set(cache_key, data, timeout=60)
        return Response(data)
    
    def get_server_metrics(self):
        """Get server metrics."""
        # [TODO: Implement using psutil or Render API]
        return {
            'cpu_usage': 45.2,
            'memory_usage': 62.7,
            'disk_usage': 38.5,
        }
    
    def get_database_metrics(self):
        """Get database metrics."""
        try:
            with connection.cursor() as cursor:
                # Get connection count
                cursor.execute("SELECT count(*) FROM pg_stat_activity;")
                connections = cursor.fetchone()[0]
                
                # Get database size
                cursor.execute("SELECT pg_database_size(current_database());")
                size = cursor.fetchone()[0]
                
                # Get cache hit rate
                cursor.execute("""
                    SELECT 
                        sum(heap_blks_hit) / (sum(heap_blks_hit) + sum(heap_blks_read)) as cache_hit_rate
                    FROM pg_statio_user_tables;
                """)
                cache_hit = cursor.fetchone()[0]
        except Exception as e:
            connections = 0
            size = 0
            cache_hit = 0
        
        return {
            'connections': connections,
            'size_mb': round(size / (1024 * 1024), 2),
            'cache_hit_rate': round(cache_hit * 100, 1) if cache_hit else 0,
        }
    
    def get_cache_metrics(self):
        """Get cache metrics."""
        try:
            from django.core.cache import cache
            if hasattr(cache, 'client'):
                client = cache.client.get_client()
                info = client.info()
                
                return {
                    'connected': True,
                    'used_memory': info.get('used_memory_human', 'N/A'),
                    'hit_rate': self._calculate_hit_rate(info),
                    'keys': info.get('db0', {}).get('keys', 0),
                }
        except Exception:
            pass
        
        return {'connected': False}
```

## Custom Dashboard Components

### Dashboard Widgets

```python
# monitoring/widgets.py
class DashboardWidget:
    """Base class for dashboard widgets."""
    
    def __init__(self, title, description, query_fn, refresh_interval=60):
        self.title = title
        self.description = description
        self.query_fn = query_fn
        self.refresh_interval = refresh_interval
        self.cache_key = f"widget:{title.lower().replace(' ', '_')}"
    
    def get_data(self):
        """Get widget data with caching."""
        data = cache.get(self.cache_key)
        if data:
            return data
        
        data = self.query_fn()
        cache.set(self.cache_key, data, timeout=self.refresh_interval)
        return data
    
    def render(self):
        """Render the widget."""
        data = self.get_data()
        return {
            'title': self.title,
            'description': self.description,
            'data': data,
            'updated_at': datetime.utcnow().isoformat() + 'Z',
        }
```

## Dashboard Refresh Strategy

| Dashboard | Refresh Rate | Cache TTL |
|-----------|--------------|-----------|
| **Health** | 1 minute | 60 seconds |
| **Business** | 5 minutes | 300 seconds |
| **Model Performance** | 1 hour | 3600 seconds |
| **Infrastructure** | 1 minute | 60 seconds |

### Implementation

```python
# monitoring/dashboards/refresh.py
from django.core.cache import cache
from django.core.management.base import BaseCommand
from datetime import datetime

class Command(BaseCommand):
    help = 'Refresh all dashboards'

    def handle(self, *args, **options):
        dashboards = [
            'health',
            'business',
            'model',
            'infrastructure',
        ]
        
        for name in dashboards:
            cache_key = f"dashboard:{name}"
            cache.delete(cache_key)
            self.stdout.write(f"Refreshed dashboard: {name}")
```
